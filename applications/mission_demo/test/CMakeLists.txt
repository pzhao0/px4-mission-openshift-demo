cmake_minimum_required(VERSION 3.8)
project(executor_regression)
set(CMAKE_CXX_STANDARD 20)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_ros2_cpp REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2 REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
set(PX4_ROS2_SOURCE_DIR "/opt/mission_ws/src/px4-ros2-interface-lib/px4_ros2_cpp"
    CACHE PATH "Pinned interface library source, for its fake registration helper")
add_executable(executor_regression executor_regression.cpp ../MissionExecutor.cpp ../DemoSession.cpp)
target_include_directories(executor_regression PRIVATE .. "${PX4_ROS2_SOURCE_DIR}")
ament_target_dependencies(executor_regression rclcpp px4_ros2_cpp tf2_ros tf2 geometry_msgs rcl_interfaces std_msgs std_srvs)
enable_testing()
add_test(NAME executor_regression COMMAND executor_regression)
set_tests_properties(executor_regression PROPERTIES
    ENVIRONMENT "ROS_DOMAIN_ID=101;ROS_LOCALHOST_ONLY=1" TIMEOUT 90)
